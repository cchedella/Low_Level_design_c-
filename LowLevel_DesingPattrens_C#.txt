#BuilderPattren

The Builder Design Pattern is a creational pattern used to construct complex objects step-by-step. It separates the representation of an object from its construction logic, allowing the same construction process to create different representations.

using System;
using System.Collections.Generic;

// 1. The Target Product (The complex object we want to build)

public class Computer
{
    public string CPU { get; }
    public int RAM { get; }
    public int Storage { get; }
    public string GraphicsCard { get; } // Optional

    // Constructor is internal/private so it can ONLY be built via the Builder

    internal Computer(string cpu, int ram, int storage, string graphicsCard)
    {
        CPU = cpu;
        RAM = ram;
        Storage = storage;
        GraphicsCard = graphicsCard;
    }

    public void DisplaySpecifications()
    {
        Console.WriteLine($"Computer Specs: CPU: {CPU}, RAM: {RAM}GB, Storage: {Storage}GB, GPU: {GraphicsCard ?? "Integrated"}");
    }
}

// 2. The Builder Interface
public interface IComputerBuilder
{
    IComputerBuilder SetCPU(string cpu);
    IComputerBuilder SetRAM(int ramGb);
    IComputerBuilder SetStorage(int storageGb);
    IComputerBuilder SetGraphicsCard(string gpu);
    Computer Build();
}

// 3. The Concrete Builder
public class ComputerBuilder : IComputerBuilder
{
    private string _cpu;
    private int _ram;
    private int _storage;
    private string _graphicsCard;

    public IComputerBuilder SetCPU(string cpu)
    {
        _cpu = cpu;
        return this; // Returns 'this' to enable method chaining
    }

    public IComputerBuilder SetRAM(int ramGb)
    {
        _ram = ramGb;
        return this;
    }

    public IComputerBuilder SetStorage(int storageGb)
    {
        _storage = storageGb;
        return this;
    }

    public IComputerBuilder SetGraphicsCard(string gpu)
    {
        _graphicsCard = gpu;
        return this;
    }

    public Computer Build()
    {
        // Validation step before creation
        if (string.IsNullOrEmpty(_cpu)) throw new InvalidOperationException("CPU is required.");
        if (_ram <= 0) throw new InvalidOperationException("Invalid RAM configuration.");

        return new Computer(_cpu, _ram, _storage, _graphicsCard);
    }
}

// 4. Usage
class Program
{
    static void Main()
    {
        // Constructing a high-end gaming PC step-by-step fluently

        Computer gamingPc = new ComputerBuilder()
            .SetCPU("Intel i9")
            .SetRAM(32)
            .SetStorage(2000)
            .SetGraphicsCard("NVIDIA RTX 4090")
            .Build();

        gamingPc.DisplaySpecifications();
    }
}

In summary, a decorator wraps an existing object and adds new functionality either before or after delegating a task to the original object.
note: this pattern is used when we want to add additional feature but the orginal code is dependent on libarires and complex we use this pattern

#Strategy Pattern


The Strategy Pattern is a behavioral design pattern that allows you to define a family of algorithms, encapsulate each one, and make them interchangeable. It lets the algorithm vary independently from the clients that use it.


// 1.The Strategy Interface

//State Pattren


State Pattren :I use the State Pattern when an object's behavior changes dynamically based on its internal status, and using traditional if/else or switch blocks would make the code unmaintainable and violate the Open/Closed Principle. Instead of keeping all state logic in one massive class, I encapsulate each state's unique behavior into its own dedicated class. The main object (the Context) simply delegates work to the current state object

Step 1 : Define the state Interface

public Interface IDocumentState
{
	void Edit(DocumnetContext context,string content);
	void SubmitForReview(DocumentContext context);
	void Approve(DocumentContext context);
	void Reject(DocumentContext context);
}

step 2 : Implement Concrete States

each class handles the bussines rules only for its specific state


//Draft State

public class DraftState  : IDocumentState
{
	public void Edit(DocumnetContext context,string content)
	{
		context.content  = content;
		Console.WriteLine("Document updated successfully");
	}

	public void SubmitForReview(DocumentContext context)
	{
		context.TrastionTo(new PendingReviewState());
		Console.WriteLine("Documnet submitted Now pending review");
	}

	public void Approve(DocumnetContext context) => Console.WriteLine("Error: Cannot approve a draft. Submit it first.");

	public void Reject(DocumnetContext context) =>Console.WriteLine("Error:Cannot reject a draft");

}


// 2. Pending review state

public class PendingReviewState : IDocumentState
{

	public void Edit(DocumnetContext context,string content) => Console.WriteLine("Error: Cannot edit while under review");

	public  void SubmitForReview(DocumnetContext context) => Console.WriteLine("Error : Already under review");

	public void Approve(DocumnetContext context)
	{
		context.TransitionTo(new ApprovedState());
		Console.WriteLine("Documnet Approve and locked");
	}

	public void Reject(DocumentContext context)
	{
		context.TransitionTo(new DraftState());
		Console.WriteLine("Documnet rejected Sent back to Draft status");
	}
}

//3. Approved State(Finale state)

public class ApprovedState:IDocumnetState
{
	public void Edit(DocumentContext context, string content) => Console.WriteLine("Error: Locked. Cannot edit an approved document.");
        public void SubmitForReview(DocumentContext context) => Console.WriteLine("Error: Already approved.");
        public void Approve(DocumentContext context) => Console.WriteLine("Error: Already approved.");
        public void Reject(DocumentContext context) => Console.WriteLine("Error: Cannot reject an already approved document.");
}

// step 3: create the context class

publc class DocumnetContext
{
	private IDocumentState _currentState;

	//public properties representing the entity data

	public string Content {get; set;}

	public DocumentContext(IDocumentState intialState)
	{
		TransitionTo(initialState);
	}

	//Method to allow states to trigger transitions

	public void TransitionTo(IDocumentState state)
	{
		_currentState = state;
	}

	//context delegates operations to the current state

	public void Edit(string content)     => _currentState.Edit(this, content);
	public void SubmitForReview()        => _currentState.SubmitForReview(this);
        public void Approve()            => _currentState.Approve(this);
        public void Reject()             => _currentState.Reject(this);
}
